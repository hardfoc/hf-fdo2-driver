47inline uint16_t Crc16Modbus(const char* data, std::size_t len) noexcept {
48 uint16_t crc = 0xFFFFU;
49 for (std::size_t i = 0; i < len; ++i) {
50 crc = static_cast<uint16_t>(crc ^ static_cast<unsigned char>(data[i]));
51 for (int bit = 0; bit < 8; ++bit) {
52 if ((crc & 1U) != 0U) {
53 crc = static_cast<uint16_t>((crc >> 1U) ^ 0xA001U);
54 } else {
55 crc = static_cast<uint16_t>(crc >> 1U);
56 }
57 }
58 }
59 return crc;
60}

72inline DriverError VerifyAndStripCrc(char* line, bool require) noexcept {
73 char* colon = std::strrchr(line, ':');
74 if (colon == nullptr) {
75 return require ? DriverError::CrcMismatch : DriverError::None;
76 }
77 const char* p = colon + 1;
78 while (*p == ' ') {
79 ++p;
80 }
81 uint32_t value = 0;
82 std::size_t nd = 0;
83 while (std::isdigit(static_cast<unsigned char>(*p)) != 0) {
84 value = value * 10U + static_cast<uint32_t>(*p - '0');
85 ++p;
86 if (++nd > 5U) {
87 return DriverError::ProtocolError;
88 }
89 }
90 if (nd == 0U || *p != '\0' || value > 0xFFFFU) {
91 return DriverError::ProtocolError;
92 }
93 const std::size_t covered = static_cast<std::size_t>(colon - line);
94 if (Crc16Modbus(line, covered) != static_cast<uint16_t>(value)) {
95 return DriverError::CrcMismatch;
96 }
97 *colon = '\0';
98 return DriverError::None;
99}

113inline DriverError ReadAsciiLine(UartT& uart, char* buf, std::size_t cap, uint32_t timeout_ms,
114 bool require_crc = false) noexcept {
115 if (cap < 2) {
116 return DriverError::InvalidParameter;
117 }
118 uint8_t tmp[512];
119 const std::size_t max_rx = sizeof(tmp) - 1U;
120 const std::size_t n = uart.read(tmp, max_rx, timeout_ms);
121 if (n == 0U) {
122 return DriverError::Timeout;
123 }
124 std::size_t pos = 0;
125 bool terminated = false;
126 for (std::size_t i = 0; i < n; ++i) {
127 const char c = static_cast<char>(tmp[i]);
128 if (c == '\r') {
129 terminated = true;
130 break;
131 }
132 if (c == '\n') {
133 continue;
134 }
135 if (pos + 1 >= cap) {
136 return DriverError::BufferOverflow;
137 }
138 buf[pos++] = c;
139 }
140 buf[pos] = '\0';
141 if (!terminated) {
142 // Data sheet §4.1: every reply ends with a single CR. Without it the line is incomplete.
143 return DriverError::ProtocolError;
144 }
145 return VerifyAndStripCrc(buf, require_crc);
146}

155inline DriverError Tokenize(const char* line, char* store, std::size_t store_cap,
156 const char* tokens[], std::size_t max_tokens,
157 std::size_t* out_count) noexcept {
158 *out_count = 0;
159 const char* p = SkipWs(line);
160 if (*p == '\0') {
161 return DriverError::ProtocolError;
162 }
163 std::size_t bi = 0;
164 while (*p != '\0' && *out_count < max_tokens) {
165 if (bi + 1U >= store_cap) {
166 return DriverError::BufferOverflow;
167 }
168 const std::size_t start = bi;
169 tokens[*out_count] = store + start;
170 while (*p != '\0' && std::isspace(static_cast<unsigned char>(*p)) == 0) {
171 if (bi + 1U >= store_cap) {
172 return DriverError::BufferOverflow;
173 }
174 store[bi++] = *p++;
175 }
176 if (bi + 1U >= store_cap) {
177 return DriverError::BufferOverflow;
178 }
179 store[bi++] = '\0';
180 ++(*out_count);
181 p = SkipWs(p);
182 }
183 if (*p != '\0') {
184 return DriverError::ProtocolError;
185 }
186 return DriverError::None;
187}
